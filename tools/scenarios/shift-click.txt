# Excel-style range extension: select a cell, shift+click another, the
# rect between them is selected. Plain click after a shift-click
# collapses. Shift+click with no selection selects the single cell.
open <PARQUET2>
wait-open
page0
click 1 0
shift-click 3 1
assert-sel-rect 1 3 0 1
assert-selected 3 1
shift-click 1 0
assert-sel-rect 1 1 0 0
clear-selection
shift-click 2 1
assert-sel-rect 2 2 1 1
click 0 0
assert-sel-rect 0 0 0 0
